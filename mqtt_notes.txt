#include "Network.h"
#include "WebSocket.h"
#include "TCP.h"

typedef enum {
    NETWORK_TYPE_TCP,
    NETWORK_TYPE_WEBSOCKET
} NetworkType;

typedef struct {
    NetworkType type;
    Network network;
} CustomNetwork;

int custom_connect(CustomNetwork* cn, char* address, int port) {
    if (cn->type == NETWORK_TYPE_TCP) {
        return tcp_connect(&cn->network, address, port);
    } else if (cn->type == NETWORK_TYPE_WEBSOCKET) {
        return websocket_connect(&cn->network, address, port);
    }
    return -1; // Invalid network type
}

int custom_read(CustomNetwork* cn, unsigned char* buffer, int len, int timeout_ms) {
    if (cn->type == NETWORK_TYPE_TCP) {
        return tcp_read(&cn->network, buffer, len, timeout_ms);
    } else if (cn->type == NETWORK_TYPE_WEBSOCKET) {
        return websocket_read(&cn->network, buffer, len, timeout_ms);
    }
    return -1; // Invalid network type
}

int custom_write(CustomNetwork* cn, unsigned char* buffer, int len, int timeout_ms) {
    if (cn->type == NETWORK_TYPE_TCP) {
        return tcp_write(&cn->network, buffer, len, timeout_ms);
    } else if (cn->type == NETWORK_TYPE_WEBSOCKET) {
        return websocket_write(&cn->network, buffer, len, timeout_ms);
    }
    return -1; // Invalid network type
}

void CustomNetworkInit(CustomNetwork* cn, NetworkType type) {
    cn->type = type;
    NetworkInit(&cn->network);
    cn->network.connect = (int (*)(Network*, char*, int))custom_connect;
    cn->network.mqttread = (int (*)(Network*, unsigned char*, int, int))custom_read;
    cn->network.mqttwrite = (int (*)(Network*, unsigned char*, int, int))custom_write;
    cn->network.disconnect = (int (*)(Network*))NetworkDisconnect;
}
